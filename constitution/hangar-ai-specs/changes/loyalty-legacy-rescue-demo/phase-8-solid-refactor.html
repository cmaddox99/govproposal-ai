// TierBenefitCalculator.java
// OCP compliant: new tiers extend,
// never modify — ENG-3.9 satisfied

public class TierBenefitCalculator {

  private static final Map<TierStatus, String>
      TIER_BENEFITS = Map.of(
    TierStatus.GENERAL,
        "Basic miles accrual, "
        + "no lounge access",
    TierStatus.GOLD,
        "1.5x miles, priority boarding, "
        + "1 upgrade",
    TierStatus.PLATINUM,
        "2x miles, lounge access, "
        + "4 upgrades",
    TierStatus.PLATINUM_PRO,
        "2.5x miles, lounge access, "
        + "8 upgrades, same-day change",
    TierStatus.EXECUTIVE_PLATINUM,
        "3x miles, Flagship Lounge, "
        + "unlimited upgrades, "
        + "dedicated agent"
  );

  public String getTierBenefitSummary(
          String tier) {
    try {
      return TIER_BENEFITS.getOrDefault(
          TierStatus.valueOf(tier),
          "Unknown tier"
      );
    } catch (IllegalArgumentException e) {
      return "Unknown tier";
    }
  }
}
// Adding MVP Gold = add ONE Map entry.
// Zero modification to existing logic.

Constitutional outcome: New tier added → add one Map.entry(). The getTierBenefitSummary method is closed for modification, open for extension via the enum and Map.
